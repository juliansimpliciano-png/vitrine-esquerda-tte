    function shuffleArray(arr) {
      const copy = [...arr];
      for (let i = copy.length - 1; i &gt; 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [copy[i], copy[j]] = [copy[j], copy[i]];
      }
      return copy;
    }

    // REGISTRO DO SERVICE WORKER E INSTALAÇÃO DO PWA
    // =========================================================================
    let deferredPrompt = null;
    const pwaInstallBanner = document.getElementById(&#39;pwaInstallBanner&#39;);
    const btnInstallPwa = document.getElementById(&#39;btnInstallPwa&#39;);
    const btnDismissPwa = document.getElementById(&#39;btnDismissPwa&#39;);

    if (&#39;serviceWorker&#39; in navigator) {
      window.addEventListener(&#39;load&#39;, () =&gt; {
        navigator.serviceWorker.register(&#39;./sw.js&#39;)
          .then((reg) =&gt; console.log(&#39;PWA ServiceWorker ativo!&#39;, reg.scope))
          .catch((err) =&gt; console.log(&#39;PWA SW error:&#39;, err));
      });
    }

    window.addEventListener(&#39;beforeinstallprompt&#39;, (e) =&gt; {
      e.preventDefault();
      deferredPrompt = e;
      if (pwaInstallBanner &amp;&amp; !sessionStorage.getItem(&#39;pwaDismissed&#39;)) {
        pwaInstallBanner.classList.remove(&#39;hidden&#39;);
      }
    });

    if (btnInstallPwa) {
      btnInstallPwa.addEventListener(&#39;click&#39;, async () =&gt; {
        if (!deferredPrompt) return;
        deferredPrompt.prompt();
        const { outcome } = await deferredPrompt.userChoice;
        if (outcome === &#39;accepted&#39;) {
          showToast(&#39;Aplicativo instalado com sucesso!&#39;, &#39;📲&#39;);
        }
        deferredPrompt = null;
        if (pwaInstallBanner) pwaInstallBanner.classList.add(&#39;hidden&#39;);
      });
    }

    if (btnDismissPwa) {
      btnDismissPwa.addEventListener(&#39;click&#39;, () =&gt; {
        sessionStorage.setItem(&#39;pwaDismissed&#39;, &#39;true&#39;);
        if (pwaInstallBanner) pwaInstallBanner.classList.add(&#39;hidden&#39;);
      });
    }
