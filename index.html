function jsonOut_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function getOrCreateMuralSheet_(ss) { var sheet = ss.getSheetByName('Mural_Oportunidades'); if (!sheet) { sheet = ss.insertSheet('Mural_Oportunidades'); sheet.appendRow([ 'ID', 'Data_Criacao', 'Data_Expiracao', 'Prazo_Dias', 'Autor_Card_ID', 'Autor_Nome', 'Autor_WhatsApp', 'Bairro_Cidade', 'Categoria', 'Subcategoria', 'Titulo', 'Descricao', 'Limite_Usos', 'Usos_Atuais', 'Resgatado_Por_IDs', 'Status' ]); sheet.setFrozenRows(1); } return sheet; } function verifyMemberByPin_(mainSheet, pinInput) { var MASTER_PIN = '191219'; var cleanPin = String(pinInput || '').trim(); if (!cleanPin) { return { ok: false, message: 'Digite seu PIN para acessar o Mural.' }; } // Acesso direto de Administrador com PIN Mestre if (cleanPin === MASTER_PIN) { return { ok: true, isMaster: true, cardId: 'ADMIN_MASTER', nome: 'Administrador da Vitrine', whatsapp: '12991839298', bairroCidade: 'Taubaté' }; } // Busca o cartão correspondente apenas pelo PIN na aba Página1 var values = mainSheet.getDataRange().getValues(); for (var i = 1; i < values.length; i++) { var row = values[i]; var rowId = String(row[0] || '').trim(); var rowNome = String(row[2] || '').trim(); var rowCidade = String(row[6] || '').trim(); var rowBairro = String(row[7] || '').trim(); var rowWhats = String(row[9] || '').replace(/\D/g, ''); var rowPin = String(row[13] || '').trim(); if (rowId && rowNome && rowPin && cleanPin === rowPin) { return { ok: true, isMaster: false, cardId: rowId, nome: rowNome, whatsapp: rowWhats, bairroCidade: rowBairro ? (rowBairro + ' • ' + rowCidade) : rowCidade }; } } return { ok: false, message: 'PIN não encontrado! Verifique seu PIN ou faça sua inscrição na Vitrine.' }; } function handleUnlockMural_(ss, mainSheet, data) { var auth = verifyMemberByPin_(mainSheet, data.pin); if (!auth.ok) { return jsonOut_({ status: 'error', message: auth.message }); } return jsonOut_({ status: 'success', member: { cardId: auth.cardId, nome: auth.nome, whatsapp: auth.whatsapp, bairroCidade: auth.bairroCidade, isMaster: auth.isMaster } }); } function handleGetMural_() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = getOrCreateMuralSheet_(ss); var values = sheet.getDataRange().getValues(); var now = new Date(); var list = []; for (var i = 1; i < values.length; i++) { var row = values[i]; var id = String(row[0] || '').trim(); if (!id) continue; var status = String(row[15] || 'Ativo').trim(); var expStr = String(row[2] || '').trim(); var limite = Number(row[12] || 0); var usos = Number(row[13] || 0); if (status !== 'Ativo') continue; var expDate = expStr ? new Date(expStr) : null; if (expDate && !isNaN(expDate.getTime())) { if (now.getTime() > expDate.getTime()) { sheet.getRange(i + 1, 16).setValue('Expirado'); continue; } } if (limite > 0 && usos >= limite) { sheet.getRange(i + 1, 16).setValue('Esgotado'); continue; } var diffDays = expDate ? Math.max(1, Math.ceil((expDate.getTime() - now.getTime()) / (1000 * 60 * 60 * 24))) : Number(row[3] || 15); list.push({ id: id, dataCriacao: String(row[1] || ''), dataExpiracao: expStr, diasRestantes: diffDays, prazoDias: Number(row[3] || 15), autorCardId: String(row[4] || ''), autor: String(row[5] || ''), whatsapp: String(row[6] || ''), bairro: String(row[7] || 'Taubaté'), categoria: String(row[8] || ''), subcategoria: String(row[9] || ''), titulo: String(row[10] || ''), descricao: String(row[11] || ''), limite: limite, usos: usos, resgatadoPorIds: String(row[14] || '').split(',').filter(Boolean) }); } return jsonOut_({ status: 'success', offers: list.reverse() }); } function handleCreateMuralOffer_(ss, mainSheet, data) { var auth = verifyMemberByPin_(mainSheet, data.pin); if (!auth.ok) { return jsonOut_({ status: 'error', message: auth.message }); } var sheet = getOrCreateMuralSheet_(ss); var values = sheet.getDataRange().getValues(); var now = new Date(); if (!auth.isMaster) { var activeByMember = 0; for (var i = 1; i < values.length; i++) { var st = String(values[i][15] || 'Ativo').trim(); var expStr = String(values[i][2] || '').trim(); var expDate = expStr ? new Date(expStr) : null; var notExpired = !expDate || isNaN(expDate.getTime()) || now.getTime() <= expDate.getTime(); if (String(values[i][4]) === String(auth.cardId) && st === 'Ativo' && notExpired) { activeByMember++; } } if (activeByMember >= 2) { return jsonOut_({ status: 'error', message: 'Limite de 2 ofertas ativas simultâneas atingido! Encerre uma oferta antiga para publicar outra.' }); } } var prazoDias = Number(data.prazoDias || 15); var expDateNew = new Date(now.getTime() + prazoDias * 24 * 60 * 60 * 1000); var offerId = 'M' + String(Date.now()); sheet.appendRow([ offerId, now.toISOString(), expDateNew.toISOString(), prazoDias, auth.cardId, auth.nome, auth.whatsapp, auth.bairroCidade, String(data.categoria || '').trim(), String(data.subcategoria || '').trim(), String(data.titulo || '').trim(), String(data.descricao || '').trim(), Number(data.limite || 0), 0, '', 'Ativo' ]); return jsonOut_({ status: 'success', message: 'Oferta publicada no Mural de Oportunidades!', offerId: offerId }); } function handleClaimMuralOffer_(ss, mainSheet, data) { var auth = verifyMemberByPin_(mainSheet, data.pin); if (!auth.ok) { return jsonOut_({ status: 'error', message: auth.message }); } var sheet = getOrCreateMuralSheet_(ss); var values = sheet.getDataRange().getValues(); var offerId = String(data.offerId || '').trim(); for (var i = 1; i < values.length; i++) { if (String(values[i][0]).trim() === offerId) { var status = String(values[i][15] || 'Ativo').trim(); var limite = Number(values[i][12] || 0); var usos = Number(values[i][13] || 0); var claimedList = String(values[i][14] || '').split(',').filter(Boolean); if (status !== 'Ativo' || (limite > 0 && usos >= limite)) { return jsonOut_({ status: 'error', message: 'Esta oferta acabou de esgotar!' }); } if (claimedList.indexOf(String(auth.cardId)) !== -1) { return jsonOut_({ status: 'error', message: 'Você já garantiu este cupom!' }); } usos += 1; claimedList.push(String(auth.cardId)); sheet.getRange(i + 1, 14).setValue(usos); sheet.getRange(i + 1, 15).setValue(claimedList.join(',')); var newStatus = (limite > 0 && usos >= limite) ? 'Esgotado' : 'Ativo'; if (newStatus === 'Esgotado') { sheet.getRange(i + 1, 16).setValue('Esgotado'); } return jsonOut_({ status: 'success', usos: usos, offerStatus: newStatus, autorWhats: String(values[i][6] || ''), titulo: String(values[i][10] || '') }); } } return jsonOut_({ status: 'error', message: 'Oferta não encontrada.' }); } function handleDeleteMuralOffer_(ss, mainSheet, data) { var auth = verifyMemberByPin_(mainSheet, data.pin); if (!auth.ok) { return jsonOut_({ status: 'error', message: auth.message }); } var sheet = getOrCreateMuralSheet_(ss); var values = sheet.getDataRange().getValues(); var offerId = String(data.offerId || '').trim(); for (var i = 1; i < values.length; i++) { if (String(values[i][0]).trim() === offerId) { var ownerId = String(values[i][4] || '').trim(); if (!auth.isMaster && ownerId !== String(auth.cardId)) { return jsonOut_({ status: 'error', message: 'Apenas o autor ou o administrador pode encerrar esta oferta.' }); } sheet.getRange(i + 1, 16).setValue('Encerrado'); return jsonOut_({ status: 'success', message: 'Oferta encerrada e removida do Mural!' }); } } return jsonOut_({ status: 'error', message: 'Oferta não encontrada.' }); }
